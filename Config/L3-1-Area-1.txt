ip access-list extended BRANCH_IN
 ! Allow return ping & TCP traffic to Branch A and Branch B ADMIN (VLAN 10)
 permit icmp any 10.1.10.0 0.0.0.255 echo-reply
 permit icmp any 10.2.10.0 0.0.0.255 echo-reply
 permit tcp any 10.1.10.0 0.0.0.255 established
 permit tcp any 10.2.10.0 0.0.0.255 established

 ! Block WIFI (VLAN 40) from internal 10.0.0.0/8 network
 deny ip 10.1.40.0 0.0.0.255 10.0.0.0 0.255.255.255

 ! Block GUICHETS (VLAN 20) & DAB (VLAN 30) from initiating traffic to ADMIN (VLAN 10)
 deny ip 10.1.20.0 0.0.0.255 10.1.10.0 0.0.0.255
 deny ip 10.1.30.0 0.0.0.255 10.1.10.0 0.0.0.255
 deny ip 10.1.20.0 0.0.0.255 10.2.10.0 0.0.0.255
 deny ip 10.1.30.0 0.0.0.255 10.2.10.0 0.0.0.255

 ! Allow Branch A & Branch B ADMIN (VLAN 10) to access Branch A Server (10.1.60.10)
 permit tcp 10.1.10.0 0.0.0.255 host 10.1.60.10 eq 21
 permit icmp 10.1.10.0 0.0.0.255 host 10.1.60.10 echo
 permit tcp 10.2.10.0 0.0.0.255 host 10.1.60.10 eq 21
 permit icmp 10.2.10.0 0.0.0.255 host 10.1.60.10 echo

 ! Block everyone else from FTP & Ping to Branch A Server (10.1.60.10)
 deny tcp any host 10.1.60.10 eq 21
 deny icmp any host 10.1.60.10 echo

 ! Permit all other traffic
 permit ip any any
 exit
